    def initialize(context):
        log.set_level('order', 'error')
        set_option('use_real_price', True)
        set_option('avoid_future_data', True)
        set_slippage(FixedSlippage(0.01))
        set_order_cost(OrderCost(open_tax=0, close_tax=0.0005, open_commission=0.0005, close_commission=0.0005, min_commission=5), type='stock')
        set_benchmark('399303.XSHE')
    
        g.information = {}
        g.drop_percent = 0.05
        g.name_cache = {}
    
        # 净值曲线动量
        g.consecutive_loss_days = 0
        g.skip_buy = False
        g.peak_value = 0
        g.drawdown_reduction = 1.0
        g.prev_day_value = 0
    
        # ML在线学习
        g.ml_features = []
        g.ml_labels = []
        g.ml_weights = None
        g.ml_pred_reduction = 1.0
        g.recent_pnls = []
        g.yesterday_buy_count = 0
        g.pending_features = None
        g.day_count = 0
    
        # ========== 统计相关 ==========
        g.condition_stats = {}          # 当前年份实际统计（每年清空）
        g.total_stats = {}              # 全历史累计实际统计（不清空）
        g.ideal_stats = {}              # 当前年份理想胜率统计（每年清空）
        g.ideal_total_stats = {}        # 全历史累计理想胜率统计
        g.current_year = None
        g.year_outputed = set()
    
        run_daily(before_market_open, time='09:10')
        run_daily(get_buy, '09:26')
        run_daily(get_close_sell, time='11:25')
        run_daily(get_close_sell, time='13:30')

    def before_market_open(context):
        y_day = context.previous_date.strftime('%Y-%m-%d')
        initial_list = prepare_stock_list(context)
        n0 = len(initial_list)
    
        g.target_list = get_stocks_with_high_increase(initial_list, y_day)
        n1 = len(g.target_list)
    
        g.target_list = filter_excessive_limit_up(g.target_list, y_day)
        n2 = len(g.target_list)
    
        g.target_list = filter_excessive_increase(g.target_list, y_day)
        n3 = len(g.target_list)
    
        g.target_list = filter_excessive_limit_days(g.target_list, y_day)
        n4 = len(g.target_list)
    
        g.target_list = filter_below_n_high(g.target_list, y_day, days=100)
        n5 = len(g.target_list)
    
        print('[选股漏斗] 基础池{} => 涨>7%({}) => 去极端涨停({}) => 去波动({}) => 去连板({}) => 近100日高({})'.format(
            n0, n1, n2, n3, n4, n5))
    
        if g.skip_buy:
            g.skip_buy = False
            print('[净值动量] 冷静期结束, 恢复交易')
    
        if g.peak_value > 0:
            current_dd = (context.portfolio.total_value / g.peak_value - 1)
            if current_dd < -0.08:
                g.drawdown_reduction = 0.5
                print('[净值动量] 回撤{:.1%}, 买入减半'.format(current_dd))
            else:
                g.drawdown_reduction = 1.0
    
        if g.ml_weights is not None and g.day_count >= 60:
            try:
                today_features = compute_ml_features(context)
                if today_features is not None:
                    score = sigmoid(np.dot(g.ml_weights, today_features))
                    if score > 0.7:
                        g.ml_pred_reduction = 0.0
                        print('[ML风控] 预测亏损概率{:.1%}, 跳过买入'.format(score))
                    elif score > 0.5:
                        g.ml_pred_reduction = 0.5
                        print('[ML风控] 预测亏损概率{:.1%}, 减半买入'.format(score))
                    else:
                        g.ml_pred_reduction = 1.0
            except:
                g.ml_pred_reduction = 1.0
        else:
            g.ml_pred_reduction = 1.0
    
        g.name_cache = {}
        if g.target_list:
            for s in g.target_list:
                try:
                    g.name_cache[s] = get_security_info(s).display_name
                except:
                    g.name_cache[s] = '未知'
    
        today_str = context.current_dt.strftime('%Y-%m-%d')
        ml_status = 'ML={}'.format('已训练' if g.ml_weights is not None else '未训练')
        dd_info = '回撤={:.1%}'.format(context.portfolio.total_value / g.peak_value - 1) if g.peak_value > 0 else '回撤=N/A'
        print('=' * 50)
        print('{} 开盘准备 | 候选{}只 | {} | {} | 减仓={:.1f}*{:.1f}'.format(
            today_str, len(g.target_list), ml_status, dd_info,
            g.drawdown_reduction, g.ml_pred_reduction))
        if g.target_list:
            stock_names = ['{}({})'.format(s, g.name_cache.get(s, '?')) for s in g.target_list[:10]]
            print('  候选: {}'.format(', '.join(stock_names)))
        print('=' * 50)
    
    
    def get_buy(context):
        if g.skip_buy:
            print('[买入] 冷静期, 不买入')
            return
        if g.ml_pred_reduction == 0.0:
            print('[买入] ML预测亏损, 跳过买入')
            return
    
        qualified_stocks = []
        current_data = get_current_data()
        y_day = context.previous_date.strftime('%Y-%m-%d')
        t_day = context.current_dt.strftime("%Y-%m-%d")
        start = t_day + ' 09:15:00'
        end = t_day + ' 09:26:00'
        DTJiner = context.portfolio.available_cash * g.drawdown_reduction * g.ml_pred_reduction
    
        if not g.target_list:
            return
    
        prev_df = get_price(
            g.target_list, end_date=y_day, frequency='daily',
            fields=['close', 'volume', 'money'], count=1, panel=False,
            fill_paused=False, skip_paused=True
        )
        prev_map = {row['code']: row for _, row in prev_df.iterrows()}
    
        val_df = get_fundamentals(
            query(valuation.code, valuation.market_cap, valuation.circulating_market_cap)
            .filter(valuation.code.in_(g.target_list)),
            date=str(y_day)[:10]
        )
        val_map = {row['code']: row for _, row in val_df.iterrows()} if not val_df.empty else {}
    
        hl_base = {s: current_data[s].high_limit / 1.1 for s in g.target_list}
    
        for s in g.target_list:
            name = g.name_cache.get(s, '未知')
            try:
                prev = prev_map.get(s)
                if prev is None:
                    continue
                avg_chg = prev['money'] / prev['volume'] / prev['close'] * 1.1 - 1
                money = prev['money']
                open_price = current_data[s].day_open
                val = val_map.get(s)
    
                if avg_chg < 0.07:
                    continue
                if open_price <= 3:
                    continue
                if val is None or val['market_cap'] < 10 or val['circulating_market_cap'] > 520:
                    continue
                if money < 1e8 or money > 15e8:
                    continue
                is_1_5 = money < 5e8
                is_5_15 = not is_1_5
            except:
                continue
    
            try:
                zyts = calculate_zyts(s, context)
                vol_data = attribute_history(s, zyts, '1d', fields=['volume'], skip_paused=True)
                if len(vol_data) < 2:
                    continue
                if vol_data['volume'][-1] <= max(vol_data['volume'][:-1]) * 0.9:
                    continue
            except:
                continue
    
            try:
                auction = get_call_auction(s, start_date=start, end_date=end, fields=['time', 'volume', 'current'])
                if auction.empty:
                    continue
                cur_ratio = auction['current'][0] / hl_base[s]
                auction_ratio = auction['volume'][0] / vol_data['volume'][-1]
    
                matched_condition = None
                for cond_name, open_lo, open_hi, auc_lo, auc_hi in CONDITION_RULES:
                    if cond_name.startswith('A') and not is_1_5:
                        continue
                    if not cond_name.startswith('A') and not is_5_15:
                        continue
                    if open_lo < cur_ratio <= open_hi and auc_lo <= auction_ratio <= auc_hi:
                        matched_condition = cond_name
                        break
    
                if matched_condition is None:
                    continue
            except:
                continue
    
            qualified_stocks.append(s)
            g.information[s] = matched_condition
            print('[竞价入选] {}({}) | {} | 竞价涨幅={:.1f}% 竞昨比={:.2f}%'.format(
                s, name, matched_condition, (cur_ratio - 1) * 100, auction_ratio * 100))
    
        print('[09:26选股] 候选{}只 => 入选{}只'.format(len(g.target_list), len(qualified_stocks)))
    
        buy_count = 0
        if qualified_stocks and context.portfolio.available_cash / context.portfolio.total_value > 0.3:
            value_per_stock = DTJiner / len(qualified_stocks)
            for s in qualified_stocks:
                price = current_data[s].last_price
                shares = int(value_per_stock / price / 100) * 100
                if shares >= 100:
                    order_value(s, value_per_stock, MarketOrderStyle(current_data[s].day_open))
                    buy_count += 1
                    name = g.name_cache.get(s, '未知')
                    print('[买入] {}({}) | 价格={:.2f} 数量={}股 条件={} 减仓={:.1f}*{:.1f}'.format(
                        s, name, price, shares, g.information.get(s, '未知'),
                        g.drawdown_reduction, g.ml_pred_reduction))
        g.yesterday_buy_count = buy_count
    
    
    def get_close_sell(context):
        y_day = context.previous_date.strftime('%Y-%m-%d')
        current_data = get_current_data()
        positions = context.portfolio.positions
    
        t = context.current_dt
        h, m = t.hour, t.minute
    
        yst_close_map = {}
        if positions:
            try:
                yst_df = get_price(
                    list(positions.keys()), end_date=y_day,
                    frequency='daily', fields=['close'], count=1,
                    panel=False, skip_paused=True
                )
                yst_close_map = dict(zip(yst_df['code'], yst_df['close']))
            except:
                pass
    
        for s in list(positions):
            if s not in g.name_cache:
                try:
                    g.name_cache[s] = get_security_info(s).display_name
                except:
                    g.name_cache[s] = '未知'
    
        if (h == 11 and m == 25) or (h == 13 and m == 30):
            for s in list(positions):
                pos = positions[s]
                last_price = current_data[s].last_price
                high_limit = current_data[s].high_limit
                avg_cost = pos.avg_cost
                closeable = pos.closeable_amount
                name = g.name_cache.get(s, '未知')
    
                # ---- 获取今日盘中最高价(用于理想胜率统计) ----
                today_high = last_price
                try:
                    mbar = attribute_history(s, 240, '1m', ['high'], skip_paused=True)
                    if len(mbar) > 0:
                        today_high = float(mbar['high'].max())
                        if today_high < last_price:
                            today_high = last_price
                except:
                    pass
    
                ideal_pct = (today_high - avg_cost) / avg_cost if avg_cost > 0 else 0
    
                try:
                    close_data2 = attribute_history(s, 4, '1d', ['close'])
                    M4 = close_data2['close'].mean()
                    MA5 = (M4 * 4 + last_price) / 5
                except:
                    continue
    
                sell_reason = None
                if closeable != 0 and last_price < high_limit and last_price > avg_cost:
                    sell_reason = '未涨停止盈'
                elif closeable != 0 and last_price < (MA5 + MA5 * 0.05):
                    sell_reason = '跌破5日线止损'
                elif closeable != 0:
                    yst_close = yst_close_map.get(s)
                    if yst_close and yst_close > 0:
                        drop_ratio = (yst_close - last_price) / yst_close
                        if drop_ratio >= g.drop_percent:
                            sell_reason = '跌幅止损'
    
                if sell_reason is not None:
                    actual_pct = (last_price - avg_cost) / avg_cost
                    cond = g.information.get(s, '未知条件')
                    print('[卖出] {}({}) | {} | 条件={} | 实际{:.2%} 理想{:.2%}(盘中高{:.2f} 成本{:.2f}) | {}:{} 持仓{}只'.format(
                        s, name, sell_reason, cond, actual_pct, ideal_pct,
                        today_high, avg_cost, h, m, len(positions)))
                    get_record_sell(context, s, sell_reason, today_high)
                    order_target_value(s, 0)
    
    
    def get_record_sell(context, stock, reason, today_high=None):
        """记录卖出统计(实际+理想胜率)"""
        try:
            pos = context.portfolio.positions.get(stock)
            if pos is None or pos.avg_cost <= 0:
                return
            current_data = get_current_data()
            price = current_data[stock].last_price
            cost = pos.avg_cost
            pct = (price - cost) / cost
    
            # 理想PnL
            ideal_pct = 0
            if today_high is not None and cost > 0:
                ideal_pct = (today_high - cost) / cost
    
            cond = g.information.get(stock, '未知条件')
    
            # ========== 更新年份实际统计 ==========
            if cond not in g.condition_stats:
                g.condition_stats[cond] = {'win': 0, 'loss': 0, 'win_pct': 0.0, 'loss_pct': 0.0}
            st = g.condition_stats[cond]
            if pct >= 0:
                st['win'] += 1
                st['win_pct'] += pct
            else:
                st['loss'] += 1
                st['loss_pct'] += pct
    
            # ========== 更新总实际统计 ==========
            if cond not in g.total_stats:
                g.total_stats[cond] = {'win': 0, 'loss': 0, 'win_pct': 0.0, 'loss_pct': 0.0}
            st_total = g.total_stats[cond]
            if pct >= 0:
                st_total['win'] += 1
                st_total['win_pct'] += pct
            else:
                st_total['loss'] += 1
                st_total['loss_pct'] += pct
    
            # ========== 更新年份理想统计 ==========
            if cond not in g.ideal_stats:
                g.ideal_stats[cond] = {'win': 0, 'loss': 0, 'win_pct': 0.0, 'loss_pct': 0.0}
            ist = g.ideal_stats[cond]
            if ideal_pct >= 0:
                ist['win'] += 1
                ist['win_pct'] += ideal_pct
            else:
                ist['loss'] += 1
                ist['loss_pct'] += ideal_pct
    
            # ========== 更新总理想统计 ==========
            if cond not in g.ideal_total_stats:
                g.ideal_total_stats[cond] = {'win': 0, 'loss': 0, 'win_pct': 0.0, 'loss_pct': 0.0}
            ist_total = g.ideal_total_stats[cond]
            if ideal_pct >= 0:
                ist_total['win'] += 1
                ist_total['win_pct'] += ideal_pct
            else:
                ist_total['loss'] += 1
                ist_total['loss_pct'] += ideal_pct
    
            # 输出当年累计统计(实际+理想均盈均亏)
            total = st['win'] + st['loss']
            ideal_total = ist['win'] + ist['loss']
            ideal_avg_win = ist['win_pct'] / ist['win'] if ist['win'] > 0 else 0
            ideal_avg_loss = ist['loss_pct'] / ist['loss'] if ist['loss'] > 0 else 0
            print('  [累计] {} | 实际: {}笔 胜率{:.1%} | 理想(盘中高): {}笔 胜率{:.1%} 均盈{:.2%} 均亏{:.2%}'.format(
                cond, total,
                st['win'] / total if total > 0 else 0,
                ideal_total,
                ist['win'] / ideal_total if ideal_total > 0 else 0,
                ideal_avg_win, ideal_avg_loss))
        except:
            pass
    
    
    def eod_stats(context):
        total_value = context.portfolio.total_value
        daily_pnl = 0
    
        g.peak_value = max(g.peak_value, total_value)
    
        if g.prev_day_value > 0:
            daily_pnl = (total_value / g.prev_day_value - 1)
            g.recent_pnls.append(daily_pnl)
            if len(g.recent_pnls) > 5:
                g.recent_pnls = g.recent_pnls[-5:]
            if daily_pnl < -0.005:
                g.consecutive_loss_days += 1
            else:
                g.consecutive_loss_days = 0
    
            if g.consecutive_loss_days >= 2:
                g.skip_buy = True
                print('[净值动量] 连亏{}天, 明日暂停买入'.format(g.consecutive_loss_days))
    
        # ML在线学习
        g.day_count += 1
    
        if g.pending_features is not None and g.prev_day_value > 0:
            label = 1.0 if daily_pnl > 0 else 0.0
            g.ml_features.append(g.pending_features)
            g.ml_labels.append(label)
            if len(g.ml_features) > 120:
                g.ml_features = g.ml_features[-120:]
                g.ml_labels = g.ml_labels[-120:]
    
        if g.day_count >= 3:
            try:
                today_f = compute_ml_features(context)
                if today_f is not None:
                    g.pending_features = today_f
            except:
                g.pending_features = None
    
        if len(g.ml_features) >= 60 and g.day_count % 5 == 0:
            try:
                train_ml_model()
            except:
                pass
    
        g.prev_day_value = total_value
    
        # 盘后摘要
        ml_status = 'ML={}'.format('已训练' if g.ml_weights is not None else '未训练')
        n_pos = len(context.portfolio.positions)
        dd_pct = (total_value / g.peak_value - 1) if g.peak_value > 0 else 0
        print('[盘后] 资产={:,.0f} 日收益={:.2%} 持仓{}只 连亏{}天 回撤={:.1%} {}'.format(
            total_value, daily_pnl, n_pos, g.consecutive_loss_days, dd_pct, ml_status))
    
        # ---------- 年度汇总与回测结束汇总 ----------
        today = context.current_dt.date()
        year = today.year
    
        if g.current_year is None:
            g.current_year = year
        elif year != g.current_year:
            if g.condition_stats:
                output_summary(g.condition_stats, g.ideal_stats, g.current_year)
            g.year_outputed.add(g.current_year)
            g.condition_stats = {}
            g.ideal_stats = {}
            g.current_year = year
    
        is_last_day = False
        try:
            next_days = get_trade_days(start_date=today, count=2)
            if len(next_days) == 1:
                is_last_day = True
        except:
            pass
    
        if is_last_day:
            if year not in g.year_outputed and g.condition_stats:
                output_summary(g.condition_stats, g.ideal_stats, year)
                g.year_outputed.add(year)
            if g.total_stats:
                output_summary(g.total_stats, g.ideal_total_stats, "全部")
    
    
    def output_summary(stats, ideal_stats, label):
        """输出指定统计字典的汇总信息（实际+理想胜率）"""
        if not stats:
            print("{}年度 竞价规则盈亏汇总：无交易记录".format(label))
            return
    
        lines = ["{}年度 竞价规则盈亏汇总（实际+理想胜率）：".format(label)]
    
        for cond, st in stats.items():
            total = st['win'] + st['loss']
            if total == 0:
                continue
            win_rate = st['win'] / total
            avg_win = st['win_pct'] / st['win'] if st['win'] > 0 else 0
            avg_loss = st['loss_pct'] / st['loss'] if st['loss'] > 0 else 0
            profit_loss_ratio = avg_win / abs(avg_loss) if avg_loss != 0 else float('inf')
    
            # 理想胜率
            ist = ideal_stats.get(cond, {'win': 0, 'loss': 0, 'win_pct': 0.0, 'loss_pct': 0.0})
            ideal_total = ist['win'] + ist['loss']
            ideal_wr = ist['win'] / ideal_total if ideal_total > 0 else 0
            ideal_avg_win = ist['win_pct'] / ist['win'] if ist['win'] > 0 else 0
            ideal_avg_loss = ist['loss_pct'] / ist['loss'] if ist['loss'] > 0 else 0
    
            lines.append(
                "  {} | 实际: 总{}笔 盈{}(均{:.2%}) 亏{}(均{:.2%}) 胜率{:.1%} 盈亏比{:.2f} | "
                "理想(盘中高): 胜率{:.1%} 均盈{:.2%} 均亏{:.2%}".format(
                    cond, total, st['win'], avg_win, st['loss'], avg_loss,
                    win_rate, profit_loss_ratio,
                    ideal_wr, ideal_avg_win, ideal_avg_loss))
        print('\n'.join(lines))
    
    
    def compute_ml_features(context):
        hs300 = '000300.XSHG'
        zz1000 = '000852.XSHG'
    
        hs300_hist = attribute_history(hs300, 60, '1d', ['close'], df=False)
        hs300_c = hs300_hist['close'][-1]
        f1 = 1.0 if hs300_c > np.mean(hs300_hist['close'][-20:]) else 0.0
        f2 = 1.0 if hs300_c > np.mean(hs300_hist['close'][-60:]) else 0.0
    
        zz1000_hist = attribute_history(zz1000, 20, '1d', ['close'], df=False)
        f3 = 1.0 if zz1000_hist['close'][-1] > np.mean(zz1000_hist['close'][-20:]) else 0.0
    
        f4 = float(len(g.target_list))
    
        rets = np.diff(hs300_hist['close'][-10:]) / hs300_hist['close'][-10:-1]
        f5 = float(np.std(rets) * np.sqrt(252))
    
        if len(g.recent_pnls) >= 3:
            f6 = float(sum(1 for p in g.recent_pnls if p > 0) / len(g.recent_pnls))

    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-np.clip(x, -20, 20)))
    
    
    def train_ml_model():
        X = np.array(g.ml_features)
        y = np.array(g.ml_labels)
        loss_mask = (y == 0)
        X_loss = X[loss_mask]
        y_loss = y[loss_mask]
        X_aug = np.vstack([X, X_loss])
        y_aug = np.concatenate([y, y_loss])
        w = np.zeros(X_aug.shape[1])
        for iteration in range(10):
            z = np.dot(X_aug, w)
            p = sigmoid(z)
            p = np.clip(p, 0.01, 0.99)
            grad = np.dot(X_aug.T, (p - y_aug))
            W = p * (1 - p)
            H = np.dot(X_aug.T * W, X_aug) + 0.01 * np.eye(X_aug.shape[1])
            try:
                w -= np.linalg.solve(H, grad)
            except:
                break
        g.ml_weights = w
        pred = sigmoid(np.dot(X, w))
        acc = np.mean((pred > 0.5) == (y > 0.5))
        print('[ML训练] 样本={} 准确率={:.1%}'.format(len(y), acc))
