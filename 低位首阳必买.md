# ========== 开盘前过滤股票 ==========
def before_open(context): 
    t_day = context.current_dt.strftime('%Y%m%d')
    y_day = context.previous_date.strftime('%Y-%m-%d')
    if not g.base_stocks:
        prepare_base_stocks(context)
    
    g.target_list = filter_first_board(g.base_stocks, y_day)
    log.info(f"[盘前] 首板筛选: {len(g.target_list)}只")
    
    g.target_list = filter_volume_price(g.target_list, y_day)
    log.info(f"[盘前] 量价过滤: {len(g.target_list)}只")
    
    g.target_list = filter_high_volatility(g.target_list, y_day)
    log.info(f"[盘前] 过滤高波动股票: {len(g.target_list)}只")
    
    g.today_pick = g.target_list.copy()
    g.pick_cache_date = t_day
    return g.today_pick

# ========== 波动过滤 ==========
def filter_high_volatility(stock_list, y_day):
    if not stock_list:
        return []
    df = get_price(stock_list, end_date=y_day, frequency='1d',
                   fields=['high', 'low'], count=5, panel=False,
                   skip_paused=True, fill_paused=False)
    if df.empty:
        return stock_list
    grp = df.groupby('code')
    chg = (grp['high'].max() - grp['low'].min()) / grp['low'].min()
    return chg[chg <= 0.2].index.tolist()
    
# ========== 量价过滤 ==========
def filter_volume_price(stock_list, y_day):
    if not stock_list:
        return []
    
    df = get_price(stock_list, end_date=y_day, frequency='1d',
                   fields=['close', 'open', 'high', 'volume', 'money'],
                   count=35, skip_paused=True, fq='pre', panel=False)
    if df.empty:
        return []
    
    candidates = []
    for code, group in df.groupby('code'):
        if len(group) < 6:
            continue
        
        group = group.sort_values('time').reset_index(drop=True)
        last = group.iloc[-1]
        d_b1 = group.iloc[-2]
        d_b2 = group.iloc[-3]
        d_b3 = group.iloc[-4]
        
        if last.money < 5e8 or last.money > 30e8:
            continue
        if d_b1.volume <= 0 or last.volume < d_b1.volume * 2:
            continue
        if d_b2.volume <= 0 or d_b1.volume > d_b2.volume * 2:
            continue
        if not (d_b1.close > d_b1.open and d_b2.close > d_b2.open):
            continue
        if d_b2.close == 0 or d_b3.close == 0:
            continue
        g1 = (d_b1.close - d_b2.close) / d_b2.close
        g2 = (d_b2.close - d_b3.close) / d_b3.close
        if g1 >= 0.05 or g2 >= 0.05:
            continue
        
        candidates.append(code)
    
    return candidates

# ========== 集合竞价过滤 ==========
def filter_auction(stock_list, y_day, t_day_str):
    if not stock_list:
        return []
    
    hist_df = get_price(stock_list, end_date=y_day, frequency='1d',
                        fields=['close', 'volume'], count=1,
                        skip_paused=True, panel=False, fill_paused=False)
    if hist_df.empty:
        return []
    
    start = f'{t_day_str} 09:15:00'
    end = f'{t_day_str} 09:26:00'
    
    qualified = []
    for stock in stock_list:
        try:
            auction = get_call_auction(stock, start_date=start, end_date=end, fields=['time', 'volume', 'current'])
            if auction.empty:
                continue
            last = auction.iloc[-1]
            hist = hist_df[hist_df['code'] == stock].iloc[0]
            if (last.volume / hist['volume'] >= 0.03 and 1.0 < last.current / hist['close'] < 1.06):
                qualified.append(stock)
        except Exception as e:
            continue
    
    return qualified

# ========== 买入股票 ==========
def get_buy(context):
    t_day = context.current_dt.strftime('%Y%m%d')
    t_day_str = context.current_dt.strftime('%Y-%m-%d')
    y_day = context.previous_date.strftime('%Y-%m-%d')
    c_data = get_current_data()
    
    holdings = [s for s, p in context.portfolio.positions.items() if p.total_amount > 0]
    if len(holdings) >= g.max_hold_count:
        return
    
    if g.pick_cache_date != t_day:
        g.today_pick = before_open(context)
        g.pick_cache_date = t_day
    
    candidates = [s for s in g.today_pick if s not in holdings]
    if not candidates:
        return
    
    qualified = filter_auction(candidates, y_day, t_day_str)
    log.info(f"[竞价过滤后]: {qualified}")
    
    if not qualified:
        return
    
    slots = g.max_hold_count - len(holdings)
    to_buy = qualified[:slots]
    
    total = context.portfolio.total_value
    max_per = total * g.max_position_pct
    per_stock = min(context.portfolio.available_cash / len(to_buy), max_per) if to_buy else 0
    
    for s in to_buy:
        try:
            op = c_data[s].day_open
            shares = int(per_stock / op / 100) * 100
            if shares >= 100:
                order_value(s, per_stock, MarketOrderStyle(op))
                log.info(f"[买入] {s} @开盘价{op:.2f}")
                g.information[s] = {'buy_date': t_day, 'buy_price': op}
        except Exception as e:
            log.info(f"[买入错误] {s}: {e}")

# ========== 止盈止损 ==========
def get_close_sell(context):
    t_day = context.current_dt.strftime("%Y%m%d")
    hold = [s for s, p in context.portfolio.positions.items()
            if p.total_amount > 0 and g.information.get(s, {}).get('buy_date') != t_day]
    if not hold:
        return
    
    metrics = g.close_sell_cache
    c_data = get_current_data()
    
    for s in hold:
        if s not in metrics:
            continue
        pos = context.portfolio.positions[s]
        cd = c_data[s]
        cp = cd.last_price
        yc = metrics[s]['y_close']
        ma5 = metrics[s]['ma5']
        dg = (cp - yc) / yc
        pp = (cp - pos.avg_cost) / pos.avg_cost
        
        if cp >= cd.high_limit * 0.985:
            continue
        
        sell, reason = False, ""
        if pp > 0.5:
            sell, reason = True, "收益>50%"
        if dg < -0.02:
            sell, reason = True, f"当日涨幅<-2%"
        elif cp < cd.day_open * 0.96:
            sell, reason = True, f"跌破开盘-4%"
        elif pp <= -0.07:
            sell, reason = True, f"亏损>7%"
        elif cp < ma5 * 0.97:
            sell, reason = True, f"跌破MA5"
        
        if sell:
            # 记录卖出信息(供盘后统计)
            buy_price = g.information.get(s, {}).get('buy_price', pos.avg_cost)
            g.today_sold.append({
                'code': s,
                'buy_price': buy_price,
                'sell_price': cp,
                'reason': reason
            })
            order_target_value(s, 0)
            log.info(f"[卖出] {s}: {reason} @{cp:.2f} 盈亏{pp*100:+.2f}%")

# ========== 每分钟止盈 ==========
def get_minute_sell(context):
    t_day_time = context.current_dt.time()
    if t_day_time < time(9, 30) or t_day_time > time(11, 25):
        return
    y_day = context.previous_date
    t_day = context.current_dt.strftime("%Y%m%d")
    hold = [s for s, p in context.portfolio.positions.items()
            if p.total_amount > 0 and g.information.get(s, {}).get('buy_date') != t_day]
    if not hold:
        return
    
    c_data = get_current_data()
    for s in hold:
        cd = c_data[s]
        cp = cd.last_price
        
        if cp >= cd.high_limit * 0.985:
            continue
        
        pos = context.portfolio.positions[s]
        buy_price = pos.avg_cost
        
        cache = g.minute_sell_cache.get(s)
        if not cache or cache['date'] != y_day:
            continue
    
        y_close = cache['y_close']
        if y_close > 0:
            yesterday_loss = (y_close - buy_price) / buy_price
            profit_ratio = (cp - buy_price) / buy_price
            if yesterday_loss < -0.02 and profit_ratio > 0.02:
                bp = g.information.get(s, {}).get('buy_price', buy_price)
                g.today_sold.append({
                    'code': s, 'buy_price': bp,
                    'sell_price': cp, 'reason': '分钟止盈-低开反弹'
                })
                order_target_value(s, 0)
                log.info(f"[分钟止盈] {s} @{cp:.2f} 盈利{profit_ratio*100:+.2f}%")
                g.today_high.pop(s, None)
                continue
        
        today_open = cd.day_open
        if today_open > 0:
            open_loss = (today_open - buy_price) / buy_price
            profit_ratio = (cp - buy_price) / buy_price
            if open_loss < -0.02 and profit_ratio > 0.02:
                bp = g.information.get(s, {}).get('buy_price', buy_price)
                g.today_sold.append({
                    'code': s, 'buy_price': bp,
                    'sell_price': cp, 'reason': '分钟止盈-开盘低开反弹'
                })
                order_target_value(s, 0)
                log.info(f"[分钟止盈] {s} @{cp:.2f} 盈利{profit_ratio*100:+.2f}%")
                g.today_high.pop(s, None)
                continue

# ========== 首板筛选 ==========
def filter_first_board(stock_list, y_day):
    df = get_price(stock_list, end_date=y_day, frequency='daily',
                   fields=['close', 'high_limit'], count=2, 
                   panel=False, fill_paused=False, skip_paused=False)
    if df.empty or len(df['time'].unique()) < 2:
        return []
    df['is_limit'] = df['close'] == df['high_limit']
    groups = df[df['is_limit']].groupby('time')['code'].apply(set)
    if len(groups) >= 2:
        return list(groups.iloc[-1] - groups.iloc[-2])
    elif len(groups) == 1:
        return list(groups.iloc[-1])
    return []
    
# ========== 盘前预计算缓存 ==========
def precompute_minute_sell_cache(context):
    y_day = context.previous_date
    pos = list(context.portfolio.positions.keys())
    if not pos:
        g.minute_sell_cache = {}
        g.close_sell_cache = {}
        return
    
    df2 = get_price(pos, end_date=y_day, frequency='1d',
                    fields=['close'], count=2, panel=False,
                    skip_paused=False, fill_paused=True)
    for s, sub in df2.groupby('code')['close']:
        if len(sub) >= 2:
            yc, pc = sub.iloc[-1], sub.iloc[-2]
            g.minute_sell_cache[s] = {
                'y_close': yc, 'prev_close': pc,
                'y_day_gain': (yc - pc) / pc if pc > 0 else 0,
                'date': y_day
            }
    
    df5 = get_price(pos, end_date=y_day, frequency='1d',
                    fields=['close'], count=5, panel=False,
                    skip_paused=False, fill_paused=True)
    g.close_sell_cache = {}
    for s, sub in df5.groupby('code')['close']:
        if len(sub) >= 2:
            g.close_sell_cache[s] = {
                'y_close': sub.iloc[-1],
                'ma5': sub.mean()
            }
    g.today_high.clear()

# ========== 基础股票池 ==========
def prepare_base_stocks(context):
    by_date = context.previous_date - timedelta(days=60)
    stock_list = list(get_all_securities(['stock'], date=by_date).index)
    current_data = get_current_data()
    stock_list = [code for code in stock_list if not (
        code.startswith(('3', '68', '4', '8', '9'))
        or current_data[code].is_st
        or current_data[code].paused
        or '退' in current_data[code].name
    )]
    g.base_stocks = get_fundamentals(
        query(valuation.code).filter(
            valuation.code.in_(stock_list),
            valuation.circulating_market_cap.between(30, 300)
        )
    )['code'].tolist()
    log.info(f"[基础股票池]: {len(g.base_stocks)}只")
    
# ========== 收盘后：理想胜率统计 ==========
def after_trading_end(context):
    """
    盘后统计理想胜率：
    - 理想胜率 = 卖出当日最高价 > 买入价 的比例(即当天有盈利机会)
    - 实际胜率 = 卖出价 > 买入价 的比例
    - 收盘胜率 = 卖出当日收盘价 > 买入价 的比例
    """
    today = context.current_dt.date()
    
    # 处理今日卖出的交易
    if g.today_sold:
        for trade in g.today_sold:
            code = trade['code']
            buy_p = trade['buy_price']
            sell_p = trade['sell_price']
            
            # 获取卖出当天的最高价和收盘价
            try:
                df = get_price(code, end_date=today, frequency='daily',
                               fields=['close', 'high'], count=1, panel=False)
                if df is not None and not df.empty:
                    day_close = float(df['close'].iloc[0])
                    day_high = float(df['high'].iloc[0])
                else:
                    day_close = sell_p
                    day_high = sell_p
            except:
                day_close = sell_p
                day_high = sell_p
            
            trade['day_close'] = day_close
            trade['day_high'] = day_high
            trade['date'] = str(today)
            
            # 计算三种收益率
            trade['ret_actual'] = (sell_p / buy_p - 1) * 100 if buy_p > 0 else 0
            trade['ret_close'] = (day_close / buy_p - 1) * 100 if buy_p > 0 else 0
            trade['ret_high'] = (day_high / buy_p - 1) * 100 if buy_p > 0 else 0
            
            g.trade_records.append(trade)
        
        # 打印今日明细
        log.info('')
        log.info('=' * 70)
        log.info(f'【今日交易明细】{today}  卖出{len(g.today_sold)}笔')
        log.info('=' * 70)
        log.info(f'  {"代码":<14}{"买入价":<9}{"卖出价":<9}{"日最高":<9}{"日收盘":<9}'
                 f'{"实际%":<9}{"理想%":<9}{"原因"}')
        log.info('  ' + '-' * 80)
        for t in g.today_sold:
            code = t['code']
            try:
                name = get_security_info(code).display_name
            except:
                name = code
            log.info(f'  {code:<14}{t["buy_price"]:<9.2f}{t["sell_price"]:<9.2f}'
                     f'{t["day_high"]:<9.2f}{t["day_close"]:<9.2f}'
                     f'{t["ret_actual"]:<+9.2f}{t["ret_high"]:<+9.2f}{t["reason"]}')
        log.info('=' * 70)
    
    # ========== 累计统计 ==========
    total = len(g.trade_records)
    if total > 0:
        win_actual = sum(1 for t in g.trade_records if t['ret_actual'] > 0)
        win_close = sum(1 for t in g.trade_records if t['ret_close'] > 0)
        win_high = sum(1 for t in g.trade_records if t['ret_high'] > 0)
        maifei = sum(1 for t in g.trade_records if (t['ret_high'] - t['ret_actual']) > 3)
        
        sum_actual = sum(t['ret_actual'] for t in g.trade_records)
        sum_high = sum(t['ret_high'] for t in g.trade_records)
        
        log.info('')
        log.info('=' * 70)
        log.info(f'【累计理想胜率统计】截至 {today}')
        log.info('=' * 70)
        log.info(f'  总交易笔数: {total}')
        log.info(f'  实际胜率(卖出价>买入价): {win_actual}/{total} = {win_actual/total*100:.1f}%')
        log.info(f'  收盘胜率(日收盘>买入价): {win_close}/{total} = {win_close/total*100:.1f}%')
        log.info(f'  理想胜率(日最高>买入价): {win_high}/{total} = {win_high/total*100:.1f}%')
        log.info(f'  卖飞率(理想-实际>3%):   {maifei}/{total} = {maifei/total*100:.1f}%')
        log.info(f'  平均实际收益: {sum_actual/total:+.2f}%')
        log.info(f'  平均理想收益: {sum_high/total:+.2f}%')
        log.info('=' * 70)
    
    # ========== 清理 ==========
    g.sold_stocks.clear()
    g.today_sold = []
    cur = set(context.portfolio.positions.keys())
    g.information = {k: v for k, v in g.information.items() if k in cur}
    g.minute_sell_cache = {k: v for k, v in g.minute_sell_cache.items() if k in cur}
    g.close_sell_cache = {k: v for k, v in g.close_sell_cache.items() if k in cur}
    g.today_high.clear()
