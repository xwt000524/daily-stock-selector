    def _get_name(stock, dt=None):
        try:
            info = get_security_info(stock, dt)
            if info:
                return info.display_name
        except:
            pass
        return stock
    
    def transform_date(date, date_type):
        if isinstance(date, str):
            str_date = date
            dt_date = dt.datetime.strptime(date, '%Y-%m-%d')
            d_date = dt_date.date()
        elif isinstance(date, dt.datetime):
            str_date = date.strftime('%Y-%m-%d')
            dt_date = date
            d_date = dt_date.date()
        elif isinstance(date, dt.date):
            str_date = date.strftime('%Y-%m-%d')
            dt_date = dt.datetime.strptime(str_date, '%Y-%m-%d')
            d_date = date
        return {'str': str_date, 'dt': dt_date, 'd': d_date}[date_type]
    
    def get_shifted_date(date, days, days_type='T'):
        d_date = transform_date(date, 'd')
        yesterday = d_date + dt.timedelta(-1)
        if days_type == 'N':
            shifted_date = yesterday + dt.timedelta(days+1)
        if days_type == 'T':
            all_trade_days = [i.strftime('%Y-%m-%d') for i in list(get_all_trade_days())]
            if str(yesterday) in all_trade_days:
                shifted_date = all_trade_days[all_trade_days.index(str(yesterday)) + days + 1]
            else:
                for i in range(100):
                    last_trade_date = yesterday - dt.timedelta(i)
                    if str(last_trade_date) in all_trade_days:
                        shifted_date = all_trade_days[all_trade_days.index(str(last_trade_date)) + days + 1]
                        break
        return str(shifted_date)
    
    def filter_st_stock(initial_list, date):
        str_date = transform_date(date, 'str')
        if get_shifted_date(str_date, 0, 'N') != get_shifted_date(str_date, 0, 'T'):
            str_date = get_shifted_date(str_date, -1, 'T')
        df = get_extras('is_st', initial_list, start_date=str_date, end_date=str_date, df=True)
        df = df.T
        df.columns = ['is_st']
        df = df[df['is_st'] == False]
        return list(df.index)
    
    def filter_kcbj_stock(initial_list):
        return [s for s in initial_list if not s.startswith(('30', '68', '4', '8'))]
    
    def filter_new_stock(initial_list, date, days=50):
        d_date = transform_date(date, 'd')
        return [s for s in initial_list if d_date - get_security_info(s).start_date > dt.timedelta(days=days)]
    
    def filter_paused_stock(initial_list, date):
        df = get_price(initial_list, end_date=date, frequency='daily', fields=['paused'], count=1, panel=False, fill_paused=True)
        df = df[df['paused'] == 0]
        return list(df.code)
    
    def filter_stock_list(stock_list, date):
        if not stock_list:
            return []
        stock_list = filter_kcbj_stock(stock_list)
        stock_list = filter_new_stock(stock_list, date)
        stock_list = filter_st_stock(stock_list, date)
        stock_list = filter_paused_stock(stock_list, date)
        return stock_list

    def normalize_concept(c_name):
        norm = re.sub(r'概念$|板块$|产业$|相关$', '', c_name)
        synonym_map = {'新能源车': '新能源汽车'}
        return synonym_map.get(norm, norm)
    
    def is_valid_concept(c_name):
        exact_blacklist = {'转融券标的', '融资融券', '国企改革', '深股通', '沪股通', '标普道琼斯A股'}
        if c_name in exact_blacklist:
            return False
        if re.search(r'\d+中报|\d+季报|\d+年报|预增|预减|预降|扭亏|盈喜|昨日|标的|股通|ST|退市|分拆上市|新股|次新|参股|摘帽|转让|信创|创投|专精特新|独角兽', c_name):
            return False
        return True
    
    def _build_stock_concept_map(date):
        """用 get_concepts() + get_concept_stocks() 反向构建 stock→concepts 映射。
        get_concept 在 jqdata 中不可用，必须通过遍历所有概念分类来获取。
        """
        date_str = str(date)
        concepts_df = get_concepts()
        if concepts_df is None or concepts_df.empty:
            print('[概念映射] get_concepts() 返回空')
            return {}
        stock_to_concepts = {}
        for _, row in concepts_df.iterrows():
            concept_id = row['id'] if 'id' in row else row.name
            concept_name = row['name']
            try:
                stocks = get_concept_stocks(concept_id, date=date_str)
                for s in stocks:
                    if s not in stock_to_concepts:
                        stock_to_concepts[s] = set()
                    stock_to_concepts[s].add(concept_name)
            except Exception:
                continue
        print(f'[概念映射] 共{len(concepts_df)}个概念分类，映射了{len(stock_to_concepts)}只股票')
        return stock_to_concepts
    
    def get_clean_concepts(stock, date, _map_cache=None):
        """获取单只股票的纯净概念集合（使用预构建映射）"""
        if _map_cache is None:
            _map_cache = _build_stock_concept_map(date)
        raw_concepts = _map_cache.get(stock, set())
        clean_set = set()
        for c_name in raw_concepts:
            norm = normalize_concept(c_name)
            if is_valid_concept(norm):
                clean_set.add(norm)
        return clean_set
    
    def get_hl_stock(initial_list, date):
        df = get_price(initial_list, end_date=date, frequency='daily', fields=['close','high_limit'], count=1, panel=False, fill_paused=False)
        df = df.dropna()
        df = df[df['close'] == df['high_limit']]
        return list(df.code)
    
    def get_continue_count_df(hl_list, date, watch_days=20):
        if not hl_list:
            return pd.DataFrame(columns=['count', 'extreme_count'])
        df = get_price(hl_list, end_date=date, frequency='daily', fields=['close','high_limit','open'], count=watch_days, panel=False, fill_paused=False)
        if df.empty:
            return pd.DataFrame(columns=['count', 'extreme_count'])
        results = []
        for stock in hl_list:
            stock_df = df[df['code'] == stock].sort_values('time', ascending=False).reset_index(drop=True)
            consecutive = 0
            extreme = 0
            for i in range(len(stock_df)):
                row = stock_df.iloc[i]
                if row['close'] == row['high_limit']:
                    consecutive += 1
                    if row['open'] >= row['high_limit'] * 0.985:
                        extreme += 1
                else:
                    break
            if consecutive > 0:
                results.append({'code': stock, 'count': consecutive, 'extreme_count': extreme})
        if results:
            return pd.DataFrame(results).set_index('code')
        else:
            return pd.DataFrame(columns=['count', 'extreme_count'])
    
    def get_hot_concept(dct, ccd, date):
        """计算当日Top5热门概念，即使结果为空也打印日志"""
        print('*******************************')
        print('[热点计算] 开始计算 Top5 热门概念...')
        
        if not dct or ccd.empty:
            print('[热点计算] 没有连板股票或概念数据，热门概念列表为空')
            return []
        
        # 一次性构建映射，后续传给 get_clean_concepts 避免重复查询
        map_cache = _build_stock_concept_map(date)
        
        concept_count = {}
        concept_contributors = {}
        for stock in dct:
            if stock not in ccd.index:
                continue
            lb_count = ccd.loc[stock, 'count']
            info = get_security_info(stock, date)
            stock_name = info.display_name if info else stock
            clean_set = get_clean_concepts(stock, date, map_cache)
            for c_name in clean_set:
                if c_name in concept_count:
                    concept_count[c_name] += lb_count
                    concept_contributors[c_name].append(f"{stock_name}({int(lb_count)}板)")
                else:
                    concept_count[c_name] = lb_count
                    concept_contributors[c_name] = [f"{stock_name}({int(lb_count)}板)"]
        
        df = pd.DataFrame({
            'concept_name': list(concept_count.keys()),
            'concept_count': list(concept_count.values()),
            'contributors': ["，".join(concept_contributors[c]) for c in concept_count.keys()]
        })
        if df.empty:
            print('[热点计算] 计算完成，但无有效概念数据（可能所有概念被清道夫过滤）')
            return []
        
        df = df.set_index('concept_name').sort_values(by=['concept_count', 'concept_name'], ascending=[False, True])
        top5 = list(df.index[:5])
        
        print('🏆 当日市场前 5 大热门主线概念：')
        for i, c in enumerate(top5):
            score = df.loc[c, 'concept_count']
            contributors = df.loc[c, 'contributors']
            print(f"  TOP {i+1}: 【{c}】 (热度得分: {score}) <- 贡献龙头: {contributors}")
        print('*******************************')
        return top5
    
    def get_market_max_boards(date):
        """获取昨日市场最高连板数"""
        all_stocks = get_all_securities('stock', date).index.tolist()
        all_stocks = filter_stock_list(all_stocks, date)
        hl_list = get_hl_stock(all_stocks, date)
        if not hl_list:
            return 0
        ccd = get_continue_count_df(hl_list, date, 20)
        if ccd.empty:
            return 0
        return int(ccd['count'].max())

    def select_pool(context):
        today = context.previous_date
        today_str = str(today)
        g.pool = {}
        g.final_pool = {}
        g.pool_filtered = False
    
        print('\n' + '=' * 60)
        print(f'【选股开始】日期: {today_str}')
        print('=' * 60)
    
        # ---- 1. 情绪指标：市场最高连板 < 4 则空仓 ----
        max_boards = get_market_max_boards(today)
        print(f'[情绪指标] 昨日市场最高连板: {max_boards}')
        if max_boards < CONFIG['min_market_boards']:
            print(f'🚫🚫🚫 [空仓信号] 最高连板{max_boards} < {CONFIG["min_market_boards"]}，今日放弃所有买入 🚫🚫🚫')
            return
    
        # ---- 2. 获取当日全部涨停股（用于计算热门概念） ----
        all_stocks = get_all_securities('stock', today).index.tolist()
        all_stocks = filter_stock_list(all_stocks, today)
        hl_all = get_hl_stock(all_stocks, today)
        print(f'[基础数据] 当日全部涨停股: {len(hl_all)}只')
    
        # 计算连板信息
        ccd_all = get_continue_count_df(hl_all, today, 20) if hl_all else pd.DataFrame()
        # 构建 stock→concepts 映射（一次性，替代不可用的 get_concept）
        concept_map = _build_stock_concept_map(today_str)
        concept_data_all = {}
        if hl_all:
            for s in hl_all:
                raw = concept_map.get(s, set())
                if raw:
                    concept_data_all[s] = {'jq_concept': [{'concept_name': n} for n in raw]}
            print(f'[概念数据] 成功获取 {len(concept_data_all)} 只股票的概念')
    
        # 计算Top5热门概念
        hot_concepts = []
        if not ccd_all.empty and concept_data_all:
            lb_stocks = [s for s in hl_all if s in ccd_all.index and ccd_all.loc[s, 'count'] >= 2]
            dct = {s: concept_data_all[s] for s in lb_stocks if s in concept_data_all and concept_data_all[s]}
            hot_concepts = get_hot_concept(dct, ccd_all, today)
        else:
            print(f'[热点计算] 跳过：ccd_all为空={ccd_all.empty}, concept_data_all有数据={bool(concept_data_all)}')
            print('[热点计算] 无连板股或概念数据，热门概念列表为空')
    
        print(f'[热点结果] 最终 Top5 热门概念: {hot_concepts if hot_concepts else "（空）"}')
    
        # ---- 3. 开始正式选股 ----
        candidates = [s for s in all_stocks if s[:2] in ('60', '00')]
        cd = get_current_data()
        candidates = [s for s in candidates if not cd[s].paused]
    
        # 预构建概念映射（供3连板热点匹配使用，复用已构建的 concept_map）
        select_map_cache = concept_map if concept_map else _build_stock_concept_map(today_str)
    
        print(f'[待选池] 主板非ST未停牌: {len(candidates)}只')
    
        n_ma = n_limit = n_consec = n_annual = n_turn_low = n_turn_high = n_slope = n_3board_miss = n_err = 0
    
        for stock in candidates:
            try:
                bar_today = get_price(stock, end_date=today, count=1,
                                      frequency='daily',
                                      fields=['close', 'high_limit', 'open', 'volume', 'money'],
                                      skip_paused=True, fq='pre')
                if bar_today is None or len(bar_today) == 0:
                    continue
                today_close = bar_today['close'].iloc[-1]
                today_hl = bar_today['high_limit'].iloc[-1]
                if today_close <= 0 or today_hl <= 0:
                    continue
    
                hist = attribute_history(stock, 250, '1d', fields=['close', 'high_limit'], skip_paused=True, fq='pre')
                if len(hist) < 30:
                    continue
                hist_c = hist['close'].values
                hist_hl = hist['high_limit'].values
                if abs(hist_c[-1] - today_close) < 0.01:
                    hc = hist_c[:-1]
                    hhl = hist_hl[:-1]
                else:
                    hc = hist_c
                    hhl = hist_hl
                if len(hc) < 29:
                    continue
    
                # MA5>MA10>MA30
                ma5 = np.mean(list(hc[-4:]) + [today_close])
                ma10 = np.mean(list(hc[-9:]) + [today_close])
                ma30 = np.mean(list(hc[-29:]) + [today_close])
                if not (ma5 > ma10 > ma30):
                    continue
                n_ma += 1
    
                # 收盘涨停
                if today_close < today_hl * 0.998:
                    continue
                n_limit += 1
    
                # 连板数
                all_c = list(hc) + [today_close]
                all_hl = list(hhl) + [today_hl]
                n_boards = 0
                for i in range(len(all_c) - 1, -1, -1):
                    if all_c[i] >= all_hl[i] * 0.998:
                        n_boards += 1
                    else:
                        break
                if n_boards > CONFIG['max_consec']:
                    n_consec += 1
                    continue
    
                # 年涨停次数
                check_len = min(250, len(all_c))
                recent_c = all_c[-check_len:]
                recent_hl = all_hl[-check_len:]
                annual_lu = sum(1 for i in range(len(recent_c)) if recent_c[i] >= recent_hl[i] * 0.998)
                if annual_lu < CONFIG['min_annual_lu']:
                    n_annual += 1
                    continue
    
                # T日换手率 (5%~15%)
                q = query(valuation.code, valuation.turnover_ratio).filter(valuation.code == stock)
                f = get_fundamentals(q, date=today)
                if f.empty:
                    continue
                t_turnover = f['turnover_ratio'].iloc[0]
                if t_turnover < CONFIG['min_turnover_t1']:
                    n_turn_low += 1
                    continue
                if t_turnover > CONFIG['max_turnover_t1']:
                    n_turn_high += 1
                    continue
    
                # 次日MA5斜率
                t1_limit = round(today_close * 1.1, 2)
                sim_ma5_t1 = np.mean(list(hc[-4:]) + [t1_limit])
                slope = (sim_ma5_t1 - ma5) / ma5 * 100
                if slope < CONFIG['min_slope']:
                    n_slope += 1
                    continue
    
                # ---- 新增：3连板必须命中Top5热门概念 ----
                if n_boards == 3:
                    if not hot_concepts:
                        n_3board_miss += 1
                        continue
                    clean_set = get_clean_concepts(stock, today, select_map_cache)
                    if not clean_set.intersection(set(hot_concepts)):
                        n_3board_miss += 1
                        continue
    
                # ---- 全部通过 ----
                g.pool[stock] = {
                    'prev_close': today_close,
                    'slope': slope,
                    'turnover': t_turnover,
                    'n_boards': n_boards,
                    'score': slope,
                }
    
            except Exception:
                n_err += 1
    
        # ---- 打印漏斗统计 ----
        print(f'\n【选股漏斗统计】')
        print(f'  MA多头: {n_ma}')
        print(f'  涨停: {n_limit}')
        print(f'  >三板剔除: {n_consec}')
        print(f'  年涨停<=10剔除: {n_annual}')
        print(f'  换手<5%剔除: {n_turn_low}')
        print(f'  换手>15%剔除: {n_turn_high}')
        print(f'  斜率<2%剔除: {n_slope}')
        print(f'  3板未命中热点剔除: {n_3board_miss}')
        print(f'  异常: {n_err}')
        print(f'✅ 最终候选池: {len(g.pool)}只')
    
        for stock in sorted(g.pool.keys(), key=lambda x: -g.pool[x]['score']):
            info = g.pool[stock]
            name = _get_name(stock, today)
            print(f'  {stock} {name} 收盘={info["prev_close"]:.2f} 斜率={info["slope"]:.1f}% 换手={info["turnover"]:.1f}% 板数={info["n_boards"]}')
    
    # ============================================================================
    # 竞价过滤 (恢复3%高开)
    # ============================================================================
    def auction_filter(context):
        if g.pool_filtered:
            return
        if not g.pool:
            print('[竞价过滤] 候选池为空，跳过竞价过滤')
            return
    
        g.pool_filtered = True
        cd = get_current_data()
        today = context.current_dt.date()
    
        g.final_pool = {}
        excluded = []
    
        prev_date = context.previous_date
        prev_vol_map = {}
        for stock in g.pool.keys():
            try:
                vol_df = get_price(stock, end_date=prev_date, count=1, frequency='daily', fields=['volume'], skip_paused=True, fq='pre')
                if vol_df is not None and len(vol_df) > 0:
                    v = vol_df['volume'].iloc[-1]
                    if v > 0:
                        prev_vol_map[stock] = v
            except:
                pass
        print(f'  [竞价诊断] 昨日成交量取到{len(prev_vol_map)}/{len(g.pool)}只')
    
        for stock, info in g.pool.items():
            try:
                cur_data = cd[stock]
                if cur_data.paused:
                    excluded.append((stock, '停牌'))
                    continue
    
                day_open = cur_data.day_open
                if day_open <= 0 or np.isnan(day_open):
                    excluded.append((stock, '开盘价无效'))
                    continue
    
                hl = cur_data.high_limit
                if hl <= 0 or np.isnan(hl) or hl > day_open * 1.5:
                    excluded.append((stock, '涨停价异常'))
                    continue
    
                prev_close = info['prev_close']
                gap_pct = (day_open - prev_close) / prev_close * 100
    
                # ---- 恢复高开>=3%过滤 ----
                if gap_pct < CONFIG['min_gap']:
                    excluded.append((stock, f'高开{gap_pct:.1f}%<3%'))
                    continue

                win = today_high > buy_price
    
                if win:

                    day_wins += 1
                else:
                    g.stat_losses += 1
                    day_losses += 1
                g.stat_total += 1
    
                name = _get_name(stock, buy_date)
                buy_str = buy_date.strftime('%m-%d') if buy_date else '?'
                tag = '胜' if win else '负'

                    'buy_date': buy_str, 'buy_price': buy_price,
                    'sell_date': today_str, 'high': today_high,
                    'diff_pct': diff_pct, 'win': win,
                    'gap_pct': gap,
                })
                print(f'[{tag}] {stock} {name} 买入={buy_str} 价={buy_price:.2f} | 次日最高={today_high:.2f} {diff_pct:+.1f}% | 高开{gap:.1f}%')
    
            except:
                pass
    
        wr = g.stat_wins * 100.0 / g.stat_total if g.stat_total > 0 else 0
